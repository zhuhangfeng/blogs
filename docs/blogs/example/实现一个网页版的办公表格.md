# 实现一个网页版的办公表格
  <br/>
  <br/>

<script setup>
  import {ref ,reactive,onMounted}from 'vue'

onMounted(()=>{
    const execl = document.getElementById('execl')
    const items = execl.querySelectorAll('.execl-item')
    let isSelect  = false
   
    items.forEach(item=>{
         item.addEventListener('mousedown',function(e){
        console.log('按下',e)
        items.forEach(item=>{
        if(!isSelect){
            item.style.backgroundColor = 'white';
        }
    })
        isSelect = true
    })
        item.addEventListener('mouseover',function(e){
            if(isSelect){
                this.style.backgroundColor ='#BDBDBD';
                console.log('移动',e)
            }
        })
        item.addEventListener('mouseup', function(e) {
            isSelect = false;
        });
    })
    
})

function handleBlur(index,e){
    console.log(index,e)
}
</script>
  <div class="execl" id="execl">
    <div class="execl-item" contenteditable="true" v-for="(item,index) in 50" @blur=handleBlur(index,$event)>{{item}}</div>
</div>

<style scoped>
  .execl{
    display: grid;

    border:1px solid #ccc;
  }

## 实现思路

- 网页版的办公表格，可以用 div+css 的方式实现，将 div 按照网格的方式排列，然后将 div 的 contenteditable 属性设置为 true，这样就可以编辑了。
- 为了实现表格的选中效果，可以给 div 添加鼠标事件，当鼠标按下时，将其他 div 的背景色设置为白色，当鼠标移动到某个 div 上时，将其背景色设置为灰色，鼠标再次按下时，将所有 div 的背景色恢复为白色。
