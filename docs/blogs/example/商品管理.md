<script>
 import { ElButton ,ElForm,ElCard,ElFormItem,ElTable,ElTableColumn,ElInput,ElCheckboxGroup,ElCheckbox} from 'element-plus'
  export default {
    name: "ProductAttrDetail",
    components: {
    ElButton,ElForm,ElCard,ElFormItem,ElTable,ElTableColumn,ElInput,ElCheckboxGroup,ElCheckbox
  },
    props: {
    },
    data() {
      return {
        selectProductAttr: [
          {name:'颜色',values:[],options:['白色','黑色','红色']},{name:'尺寸',values:[],options:['S','M','L','XL','XXL']}
        ],
          skuStockList:[]
      }
    },
    computed: {
    },
    created() {
    },
    watch: {
    },
    methods: {
      handleChange(value){
        let colors = this.selectProductAttr[0].values;
  let sizes = this.selectProductAttr[1].values;

  // 先备份当前的 skuStockList
  let oldSkuStockList = JSON.parse(JSON.stringify(this.skuStockList));

  this.skuStockList = this.merge(colors.slice(), sizes);

  // 检查如果新生成的列表比旧列表短，可能有取消选中的情况
    // 遍历旧列表，找到不在新列表中的项，恢复其数据到新列表
    oldSkuStockList.forEach(item => {
      for(let i=0;i<this.skuStockList.length;i++){
        if(item.id == this.skuStockList[i].id){
          this.skuStockList[i] = item
        }
      }
    });
      },
      handleRefreshProductSkuList(){
        // 合并参数
        let colors = this.selectProductAttr[0].values
        let sizes = this.selectProductAttr[1].values
        this.skuStockList = this.merge(colors.slice(),sizes)
      },
       merge(colors,sizes,result=[],num=0){
        if(colors.length === 0){
          return result
        }
        
        let color = colors.shift()
        for(const size of sizes){
          result.push({color:color,size:size,piace:'1',num:'1',id:num++})
        }
        return this.merge(colors,sizes,result,num)
      },
      handleInput(val){
        console.log(val)
        // 缓存数据
        sessionStorage.setItem('myData', JSON.stringify(this.skuStockList))
        console.log(JSON.parse(sessionStorage.getItem('myData')))
      }
    }
  }
</script>

## 实现思路

  - svg标签定义了一个可缩放矢量图形区域。

  - defs元素主要用于定义可重复使用的图形元素和其他资源，可以在 defs 中定义复杂的路径、图形样式等

  - path元素定义了路径，d="M-160 44c30 0 58-18 88-18s 58 18 88 18 58-18 88-18 58 18 88 18 v44h-352z", M-160 44 表示移动到坐标点 (-160, 44)，这是路径的起始点。c30 0 58-18 88-18s 58 18 88 18 58-18 88-18 58 18 88 18 ,这部分是一系列三次贝塞尔曲线（Cubic Bezier Curve）的控制指令，c 代表曲线（Curve）。每个控制点由四个数字表示，分别为 x 和 y 坐标。这些控制点决定了曲线的形状。v44 表示绘制一条垂直的直线到纵坐标为 44 的位置。h-352 表示绘制一条水平的直线向左移动 352 个单位。

  - g是 SVG 中的一个分组元素，可以将相关的图形元素放置在一个组内，方便对它们进行统一的操作和管理，比如移动、旋转、缩放整个组的元素。

  - use标签，它们分别引用了同一个波浪形状图形元素#gentle-wave，控制不同透明度、y轴不同值错开形成波浪效果。

  - 最后使用keyframes设置波浪形状图形向右平移，使用nth-child伪类选择器，选择父级下面的每一个子元素，设置其不同动画延迟时间和动画结束时间。
